# Configuración base
$usuario = $env:USERNAME
$server  = "master-vera"
$dominio = "CAMPUSVERA"
$shareRW = "GestorDocumental"

# Limpiar unidades previas
foreach ($unidad in "X","Y","Z") {
    net use "${unidad}:" /delete /y 2>$null | Out-Null
}

# Obtener grupos del usuario actual
$identity      = [System.Security.Principal.WindowsIdentity]::GetCurrent()
$gruposUsuario = $identity.Groups | ForEach-Object {
    try   { $_.Translate([System.Security.Principal.NTAccount]).Value }
    catch { $_.Value }
}

# Mapeo grupo -> departamento
$mapeo = @{
    "Grupo-GG-Prensa"         = "Prensa"
    "Grupo-GG-Administracion" = "Administración"
    "Grupo-GG-Informatica"    = "Informática"
    "Grupo-GG-Rectorado"      = "Rectorado"
}

# Determinar departamento del usuario
$departamento = $null
foreach ($grupo in $gruposUsuario) {
    foreach ($clave in $mapeo.Keys) {
        if ($grupo -like "*$clave*") {
            $departamento = $mapeo[$clave]
            break
        }
    }
    if ($departamento) { break }
}

# Mapear unidades (los permisos NTFS del servidor controlan el acceso)
$rutaUNC = "\\$server\$shareRW"
net use "X:" $rutaUNC /persistent:no 2>&1 | Out-Null
net use "Y:" $rutaUNC /persistent:no 2>&1 | Out-Null
net use "Z:" $rutaUNC /persistent:no 2>&1 | Out-Null

if ($LASTEXITCODE -ne 0) {
    Write-Warning "Error al mapear. Comprueba que \\$server\$shareRW es accesible."
}

# Mensaje de bienvenida
if ($departamento) {
    $msg = "Bienvenido/a, $usuario. Tus unidades X:, Y: y Z: apuntan al servidor de archivos. Tu carpeta de departamento es: $departamento."
} else {
    $msg = "Bienvenido/a, $usuario. Unidades mapeadas. No se ha detectado grupo de departamento asignado."
}

msg $env:USERNAME $msg
